using Libreria_restaurante.Entidades;
using Libreria_restaurante.Interfaces;
using Microsoft.EntityFrameworkCore;
using System;
using System.Collections.Generic;
using System.Text;

namespace Libreria_restaurante.Implementaciones
{
    public class CategoriasAplicacion : ICategoriasAplicacion
    {
        private IConexion IConexion;

        public CategoriasAplicacion(IConexion IConexion)
        {
            this.IConexion = IConexion;
        }

        public Categorias Insertar(Categorias entidad)
        {
            if (entidad.id != 0)
                throw new Exception("Ya se guardo");

            if (Validar(entidad))
                throw new Exception("Falta informacion");

            this.IConexion.Categorias!.Add(entidad);
            this.IConexion.SaveChanges();

            var audit = new Auditorias
            {
                tabla = "Categorias",
                accion = "Insertar",
                fecha = DateTime.Now,
                datosAnteriores = "",
                datosNuevos = "id: " + entidad.id + ", nombre: " + entidad.nombre
            };
            this.IConexion.Auditorias!.Add(audit);
            this.IConexion.SaveChanges();
            return entidad;
        }

        public Categorias Actualizar(Categorias entidad)
        {
            if (entidad.id == 0)
                throw new Exception("No se ha guardo");

            if (Validar(entidad))
                throw new Exception("Falta informacion");

            var anterior = this.IConexion.Categorias!.AsNoTracking().FirstOrDefault(x => x.id == entidad.id);
            if (anterior == null)
                throw new Exception("No existe");

            var entry = this.IConexion.Entry<Categorias>(entidad);
            entry.State = EntityState.Modified;

            var audit = new Auditorias
            {
                tabla = "Categorias",
                accion = "Actualizar",
                fecha = DateTime.Now,
                datosAnteriores = "id: " + anterior.id + ", nombre: " + anterior.nombre,
                datosNuevos = "id: " + entidad.id + ", nombre: " + entidad.nombre
            };
            this.IConexion.Auditorias!.Add(audit);
            this.IConexion.SaveChanges();
            return entidad;
        }

        public List<Categorias> Consultar()
        {
            return this.IConexion.Categorias!.ToList();
        }

        public List<Categorias> Buscar(Categorias entidad)
        {
            return this.IConexion.Categorias!
                .Where(x => x.nombre!.Contains(entidad.nombre!))
                .ToList();
        }

        public Categorias Borrar(Categorias entidad)
        {
            if (entidad.id == 0)
                throw new Exception("No se ha guardo");

            var itemDb = this.IConexion.Categorias!.FirstOrDefault(x => x.id == entidad.id);
            if (itemDb == null)
                throw new Exception("La categoría no existe");

            if (this.IConexion.Productos!.Any(p => p.categoria == itemDb.id))
                throw new Exception("No se puede eliminar la categoría porque tiene productos asociados.");

            this.IConexion.Categorias!.Remove(itemDb);

            var audit = new Auditorias
            {
                tabla = "Categorias",
                accion = "Borrar",
                fecha = DateTime.Now,
                datosAnteriores = "id: " + itemDb.id + ", nombre: " + itemDb.nombre,
                datosNuevos = ""
            };
            this.IConexion.Auditorias!.Add(audit);
            this.IConexion.SaveChanges();
            return itemDb;
        }

        private bool Validar(Categorias entidad)
        {
            return string.IsNullOrEmpty(entidad.nombre);
        }
    }
}

        {
            if (entidad.id != 0)
                throw new Exception("Ya se guardo");

            if (entidad.pedido == 0 || entidad.producto == 0 || entidad.cantidad <= 0)
                throw new Exception("Falta informacion");

       
            var productoDb = this.IConexion.Productos!.FirstOrDefault(p => p.id == entidad.producto);
            if (productoDb == null)
                throw new Exception("El producto seleccionado no existe.");

            if (productoDb.precio <= 0)
                throw new Exception($"El producto '{productoDb.nombre}' no tiene un precio configurado.");

            entidad.subtotal = Math.Round(productoDb.precio * entidad.cantidad, 2);

            
            var recetasIds = this.IConexion.Recetas!
                .Where(r => r.producto == entidad.producto)
                .Select(r => r.id)
                .ToList();

                .Where(ing => recetasIds.Contains(ing.receta))
                .ToList();

            foreach (var ing in ingredientes)
            {
                var inventario = this.IConexion.Inventarios!
                    .FirstOrDefault(inv => inv.ingrediente == ing.id);

                if (inventario != null)
                {
                    if (inventario.stock_actual < entidad.cantidad)
                        throw new Exception($"Stock insuficiente para el ingrediente: {ing.nombre}");

                    inventario.stock_actual -= entidad.cantidad;
                    inventario.fecha_actualizacion = DateTime.Now;
                    this.IConexion.Entry(inventario).State = EntityState.Modified;
                }
            }

            this.IConexion.Detalles_Pedidos!.Add(entidad);
            this.IConexion.SaveChanges();

            var audit = new Auditorias
            {

                accion = "Insertar",

        {
            this.IConexion = IConexion;
        }

        public Facturas Insertar(Facturas entidad)
        {
            if (entidad.id != 0)
                throw new Exception("Ya se guardo");

            if (entidad.pedido == 0)
                throw new Exception("Debe seleccionar un pedido");

            var pedido = this.IConexion.Pedidos!.FirstOrDefault(p => p.id == entidad.pedido);
            if (pedido == null)
                throw new Exception("El pedido no existe");

            // Evitar facturas duplicadas para el mismo pedido
            var facturaExistente = this.IConexion.Facturas!.FirstOrDefault(f => f.pedido == entidad.pedido);
            if (facturaExistente != null)
                throw new Exception($"El pedido #{entidad.pedido} ya tiene una factura generada ({facturaExistente.numero}).");

            if (string.IsNullOrEmpty(entidad.numero))
                entidad.numero = $"FAC-{DateTime.Now:yyyyMMddHHmmss}";

            if (entidad.fecha_emision == default)
                entidad.fecha_emision = DateTime.Now;

            if (entidad.subtotal <= 0)
            {
                var detalles = this.IConexion.Detalles_Pedidos!
                    .Where(d => d.pedido == entidad.pedido)
                    .ToList();
                entidad.subtotal = detalles.Sum(d => d.subtotal);
            }

            if (entidad.total <= 0)
            {
                // IVA legal del 19%
                decimal iva = Math.Round(entidad.subtotal * 0.19m, 2);
                entidad.total = entidad.subtotal + iva;
            }

            // Desvincular navegaciones para prevenir problemas de ChangeTracker en EF Core
            entidad._pedido = null;
            entidad.Pagos = new();

            // La mesa NO se libera y el pedido NO se cancela aquí. Se liberará y cancelará una vez completado el pago.
            this.IConexion.Facturas!.Add(entidad);
            this.IConexion.SaveChanges();

                accion = "Insertar",
                fecha = DateTime.Now,

            };

using Libreria_restaurante.Entidades;
using Libreria_restaurante.Interfaces;
using Microsoft.EntityFrameworkCore;
using System;
using System.Collections.Generic;
using System.Linq;

namespace Libreria_restaurante.Implementaciones
{
    public class PagosAplicacion : IPagosAplicacion
    {
        private IConexion IConexion;

        public PagosAplicacion(IConexion IConexion)
        {
            this.IConexion = IConexion;
        }

        public Pagos Insertar(Pagos entidad)
        {
            if (entidad.id != 0)
                throw new Exception("Ya se guardo");

            if (entidad.factura == 0)
                throw new Exception("Debe seleccionar una factura");

            if (entidad.metodoPago == 0)
                throw new Exception("Debe seleccionar un método de pago");

            if (entidad.monto <= 0)
                throw new Exception("El monto a pagar debe ser mayor a cero");

            if (string.IsNullOrEmpty(entidad.referencia))
                entidad.referencia = $"PAG-{DateTime.Now:yyyyMMddHHmmss}";

            if (string.IsNullOrEmpty(entidad.estado))
                entidad.estado = "Aprobado";

            if (entidad.fecha == default)
                entidad.fecha = DateTime.Now;

            entidad._factura = null;
            entidad._metodoPago = null;

            var factura = this.IConexion.Facturas!.FirstOrDefault(f => f.id == entidad.factura);
            if (factura == null)
                throw new Exception("La factura no existe");

            // Validar que la factura no esté ya cancelada o saldada
            if (factura.total > 0)
            {
                var pagosPrevios = this.IConexion.Pagos!.Where(p => p.factura == factura.id).ToList();
                decimal acumulado = pagosPrevios.Sum(p => p.monto);
                if (acumulado >= factura.total)
                    throw new Exception("La factura ya ha sido pagada en su totalidad.");
            }

            this.IConexion.Pagos!.Add(entidad);
            this.IConexion.SaveChanges();

            // Sincronización de Pedido y Mesa: cuando la factura se paga por completo
            if (factura.pedido > 0)
            {
                var pedido = this.IConexion.Pedidos!.FirstOrDefault(p => p.id == factura.pedido);
                if (pedido != null)
                {
                    var todosPagos = this.IConexion.Pagos!.Where(p => p.factura == factura.id).ToList();
                    decimal totalPagado = todosPagos.Sum(p => p.monto);

                    if (factura.total <= 0 || totalPagado >= factura.total)
                    {
                        pedido.estado = "Cancelado";
                        this.IConexion.Entry(pedido).State = EntityState.Modified;

                        var mesa = this.IConexion.Mesas!.FirstOrDefault(m => m.id == pedido.mesa);
                        if (mesa != null)
                        {
                            mesa.estado = "Disponible";
                            this.IConexion.Entry(mesa).State = EntityState.Modified;
                        }
                        this.IConexion.SaveChanges();
                    }
                }
            }

            var audit = new Auditorias
            {
                tabla = "Pagos",
                accion = "Insertar",
                fecha = DateTime.Now,
                datosAnteriores = "",
                datosNuevos = "id: " + entidad.id + ", referencia: " + entidad.referencia
            };
            this.IConexion.Auditorias!.Add(audit);
            this.IConexion.SaveChanges();
            return entidad;
        }

        public Pagos Actualizar(Pagos entidad)
        {
            if (entidad.id == 0)
                throw new Exception("No se ha guardo");

            if (Validar(entidad))
                throw new Exception("Falta informacion");

            var anterior = this.IConexion.Pagos!.AsNoTracking().FirstOrDefault(x => x.id == entidad.id);
            if (anterior == null)
                throw new Exception("No existe");

            entidad._factura = null;
            entidad._metodoPago = null;

            var entry = this.IConexion.Entry<Pagos>(entidad);
            entry.State = EntityState.Modified;

            var audit = new Auditorias
            {
                tabla = "Pagos",
                accion = "Actualizar",

            if (entidad.id == 0)
                throw new Exception("No se ha guardo");

            if (entidad.mesa == 0)
                throw new Exception("Debe seleccionar una mesa");

            var anterior = this.IConexion.Pedidos!.AsNoTracking().FirstOrDefault(x => x.id == entidad.id);
            if (anterior == null)
                throw new Exception("No existe");

            var pedidoDb = this.IConexion.Pedidos!.FirstOrDefault(x => x.id == entidad.id);
            if (pedidoDb == null)
                throw new Exception("No existe");

            pedidoDb.estado = !string.IsNullOrEmpty(entidad.estado) ? entidad.estado : anterior.estado;
            pedidoDb.numero_personas = entidad.numero_personas > 0 ? entidad.numero_personas : pedidoDb.numero_personas;
            pedidoDb.fecha_hora = entidad.fecha_hora != default ? entidad.fecha_hora : pedidoDb.fecha_hora;
            if (entidad.cliente != 0) pedidoDb.cliente = entidad.cliente;
            if (entidad.mesero != 0) pedidoDb.mesero = entidad.mesero;
            if (entidad.mesa != 0) pedidoDb.mesa = entidad.mesa;

            this.IConexion.Entry(pedidoDb).State = EntityState.Modified;

            var mesa = this.IConexion.Mesas!.Find(pedidoDb.mesa);
            if (mesa != null && !string.IsNullOrEmpty(pedidoDb.estado))
            {
                if (pedidoDb.estado.Equals("Pagado", StringComparison.OrdinalIgnoreCase) ||
                    pedidoDb.estado.Equals("Cancelado", StringComparison.OrdinalIgnoreCase) ||
                    pedidoDb.estado.Equals("Finalizado", StringComparison.OrdinalIgnoreCase))
                {
                    mesa.estado = "Disponible";
                }
                else
                {
                    mesa.estado = "Ocupada";
                }
            }

            var audit = new Auditorias
            {
                tabla = "Pedidos",
                accion = "Actualizar",
                fecha = DateTime.Now,
                datosAnteriores = "id: " + anterior.id + ", estado: " + anterior.estado,
                datosNuevos = "id: " + pedidoDb.id + ", estado: " + pedidoDb.estado
            };
            this.IConexion.Auditorias!.Add(audit);
            this.IConexion.SaveChanges();
            return pedidoDb;
        }

                .Where(x => x.estado!.Contains(entidad.estado!))
                .ToList();
        }

        public Pedidos Borrar(Pedidos entidad)
        {
            if (entidad.id == 0)
                throw new Exception("No se ha guardo");

            // Buscar la entidad rastreada directamente desde la BD para evitar conflictos de claves/navegación
            var pedidoDb = this.IConexion.Pedidos!.FirstOrDefault(p => p.id == entidad.id);
            if (pedidoDb == null)
                throw new Exception("El pedido no existe");

            // 1. Si tiene factura asociada, no permitir borrar sin anular o eliminar factura
            if (this.IConexion.Facturas!.Any(f => f.pedido == pedidoDb.id))
                throw new Exception("No se puede borrar un pedido que ya tiene factura generada.");

            // 2. Eliminar detalles del pedido y devolver stock al inventario
            var detalles = this.IConexion.Detalles_Pedidos!.Where(d => d.pedido == pedidoDb.id).ToList();
            foreach (var det in detalles)
            {
                var recetasIds = this.IConexion.Recetas!
                    .Where(r => r.producto == det.producto)
                    .Select(r => r.id)
                    .ToList();

                var ingredientes = this.IConexion.Ingredientes!
                    .Where(ing => recetasIds.Contains(ing.receta))
                    .ToList();

                foreach (var ing in ingredientes)
                {
                    var inv = this.IConexion.Inventarios!.FirstOrDefault(i => i.ingrediente == ing.id);
                    if (inv != null)
                    {
                        inv.stock_actual += det.cantidad;
                        inv.fecha_actualizacion = DateTime.Now;
                        this.IConexion.Entry(inv).State = EntityState.Modified;
                    }
                }

                this.IConexion.Detalles_Pedidos!.Remove(det);
            }

            // 3. Liberar la mesa
            var mesa = this.IConexion.Mesas!.FirstOrDefault(m => m.id == pedidoDb.mesa);
            if (mesa != null)
            {
                mesa.estado = "Disponible";
                this.IConexion.Entry(mesa).State = EntityState.Modified;
            }

            string infoAnterior = $"id: {pedidoDb.id}, estado: {pedidoDb.estado}";
            this.IConexion.Pedidos!.Remove(pedidoDb);

            var audit = new Auditorias
            {
                tabla = "Pedidos",
                accion = "Borrar",
                fecha = DateTime.Now,
                datosAnteriores = infoAnterior,
                datosNuevos = ""
            };
            this.IConexion.Auditorias!.Add(audit);
            this.IConexion.SaveChanges();
            return pedidoDb;
        }
